156 int hex_value(char c) {
157 if (c >= '0' && c <= '9') return c - '0';
158 if (c >= 'a' && c <= 'f') return 10 + (c - 'a');
159 if (c >= 'A' && c <= 'F') return 10 + (c - 'A');
160 return -1;

178int parse_hex_string(const char *hex, uint8_t *out, size_t *out_len) {
179 size_t len = strlen(hex);
180 if (len == 0) {
181 printf("bruh\n");
182 return -1;
183 }
184 // faccio tutte le vaiidazioni del caso
185
186 // !-- numero di nibble pari --!
187 if (len % 2 != 0) {
188 printf("n nibble dispari (%zu)\n", len);
189 return -1;
190 }
191 // !-- almeno 14 byte totali; --!
192 size_t byte_len = len / 2; // un byte = 2 char hex
193 if (byte_len < MIN_FRAME_BYTES) {
194 printf("Frame too short (%zu byte, min %u)\n", byte_len, MIN_FRAME_BYTES);
195 return -1;
196 }
197 // !-- payload ≤ 1500 byte (quindi len ≥ 14 e len-14 ≤ 1500). --!
198 if (byte_len > MAX_FRAME_BYTES) {
199 printf("Frame too long (%zu byte, max %u)\n", byte_len, MAX_FRAME_BYTES);
200 return -1;
201 }
202
203 /* scorro di 2 posizioni e prendo i 2 hex char per trasformarli in un byte (1 byte = 2 hex char = 2 nibbl3) */
204 for (size_t i = 0; i < byte_len; ++i) {
205 int n1 = hex_value(hex[2 * i]); // 2*0 = 0, 2*1 = 2
206 int n2 = hex_value(hex[2 * i + 1]); // 2*0+1 = 1, 2*1+1 = 3
207
208 out[i] = (uint8_t)((n1 << 4) | n2); // sposto n1 di un nibble a sinistra e faccio or con n2 (unione )
209 }
210
211 *out_len = byte_len;
212 return 0;
213}

225void fill_frame(const uint8_t *raw, size_t len, EthFrame *frame) {
226 for (size_t i = 0; i < ETH_MAC_LEN; ++i) { // prendo i 6 byte del MAC
227 frame->dst_mac[i] = raw[i];
228 }
229 for (size_t i = 0; i < ETH_MAC_LEN; ++i) { // prendo 6 byte dell'altro mac
230 frame->src_mac[i] = raw[ETH_MAC_LEN + i];
231 }
232 // 2 byte ether type
233 frame->ether_type = (uint16_t)((raw[12] << 8) | raw[13]); // cast esplicito a 16 bit unsigned | predo il byte 12 (6+6+1) e shift di 8 or byte 13
234 frame->payload_len = len - MIN_FRAME_BYTES; // len payload = tot len - 14 byte (6+6+2)
235 frame->total_len = len; // len totale del frame
236
237 for (size_t i = 0; i < frame->payload_len; ++i) {
238 frame->payload[i] = raw[MIN_FRAME_BYTES + i]; // offset di 14 poi copia bit a bit
239 }
240}

251 const char *describe_mac(const uint8_t mac[6]) {
252 int is_broadcast = 1;
253 for (size_t i = 0; i < 6; ++i) {
254 if (mac[i] != 0xFF) {
255 is_broadcast = 0; // se trovo un byte nn FF no broadcast
256 break;
257 }
258 }
259 if (is_broadcast) {
260 return "Broadcast";
261 }
262
263 int is_multicast = (mac[0] & 0x01) != 0;
264 int is_local = (mac[0] & 0x02) != 0;
265 if (is_multicast) {
266 return is_local ? "multicast locale" : "multicast universale";
267 }
268 return is_local ? "unicast locale" : "unicast universale"; // operatore ternario per dio
269}
